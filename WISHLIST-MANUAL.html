Wishlist Integration Manual

Add wishlist to a Shopify theme and connect it to the app API.

API base URL: https://personalize.salty.co.in

Guest users save products in the browser. Logged-in customers sync the same list to the app.

Where to put files in the theme

Create these files in the theme, then apply the patches in steps 10–13.

Step 1 — assets/wishlist.js

Create this file in the theme assets folder.

assets/wishlist.js 
(() => {
  const STORAGE_KEY = "shopify_wishlist";
  const API_BASE = () =>
    String(window.WISHLIST_API_URL || "https://personalize.salty.co.in").replace(
      /\/$/,
      "",
    );

  const customerId = () => {
    const id = window.SHOPIFY_CUSTOMER_ID;
    if (id == null || id === "" || id === "null") return null;
    return String(id);
  };

  const uniqueHandles = (handles) => {
    const seen = new Set();
    const result = [];
    for (const value of handles || []) {
      const handle = String(value || "").trim();
      if (!handle || seen.has(handle)) continue;
      seen.add(handle);
      result.push(handle);
    }
    return result;
  };

  const readLocal = () => {
    try {
      const raw = localStorage.getItem(STORAGE_KEY);
      if (!raw) return [];
      const parsed = JSON.parse(raw);
      return uniqueHandles(Array.isArray(parsed) ? parsed : []);
    } catch {
      return [];
    }
  };

  const writeLocal = (handles) => {
    const next = uniqueHandles(handles);
    localStorage.setItem(STORAGE_KEY, JSON.stringify(next));
    return next;
  };

  const parseApiHandles = (payload) => {
    if (!payload) return [];
    if (Array.isArray(payload)) return uniqueHandles(payload);
    if (Array.isArray(payload.handles)) return uniqueHandles(payload.handles);
    if (typeof payload === "string") return uniqueHandles(payload.split(","));
    if (typeof payload.handles === "string") {
      return uniqueHandles(payload.handles.split(","));
    }
    return [];
  };

  const setCount = (count) => {
    document.querySelectorAll(".wishlist-count").forEach((el) => {
      el.textContent = String(count);
      el.setAttribute("data-count", String(count));
    });
  };

  const updateIcons = (handles = readLocal()) => {
    document.querySelectorAll(".wishlist-btn").forEach((btn) => {
      const handle = (btn.dataset.handle || "").trim();
      if (!handle) return;
      const active = handles.includes(handle);
      btn.classList.toggle("active", active);
      btn.setAttribute("aria-pressed", active ? "true" : "false");
      btn.setAttribute(
        "aria-label",
        active ? "Remove from wishlist" : "Add to wishlist",
      );
    });
    setCount(handles.length);
  };

  const saveRemote = async (handles) => {
    const id = customerId();
    if (!id) return;

    try {
      await fetch(`${API_BASE()}/api/wishlist`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          actionType: handles.length === 0 ? "remove" : "save",
          productHandle: handles.join(","),
          customerId: id,
        }),
      });
    } catch (error) {
      console.error("Wishlist save failed:", error);
    }
  };

  const fetchRemote = async () => {
    const id = customerId();
    if (!id) return [];

    try {
      const response = await fetch(
        `${API_BASE()}/api/wishlist-list?customerId=${encodeURIComponent(id)}`,
      );
      if (!response.ok) return [];
      return parseApiHandles(await response.json());
    } catch (error) {
      console.error("Wishlist list failed:", error);
      return [];
    }
  };

  const mergeWishlists = (localHandles, remoteHandles) =>
    uniqueHandles([...(remoteHandles || []), ...(localHandles || [])]);

  const persist = async (handles) => {
    const next = writeLocal(handles);
    updateIcons(next);
    await saveRemote(next);
    return next;
  };

  const showEmpty = (container) => {
    const text = container?.dataset.emptyText || "Your wishlist is empty.";
    if (!container) return;
    container.innerHTML = `<p class="wishlist-empty-text">${text}</p>`;
    document.getElementById("MainContent")?.classList.add("wishlist-empty");
  };

  const loadWishlistPage = async (handles) => {
    const container = document.getElementById("wishlist-container");
    if (!container) return;

    const list = uniqueHandles(handles);
    container.innerHTML = "";

    if (!list.length) {
      showEmpty(container);
      return;
    }

    container.classList.add("is-loading");
    document.getElementById("MainContent")?.classList.remove("wishlist-empty");

    const cards = await Promise.all(
      list.map(async (handle) => {
        try {
          const response = await fetch(
            `/products/${encodeURIComponent(handle)}?view=wishlist-card`,
          );
          if (!response.ok) return null;
          const html = await response.text();
          const temp = document.createElement("div");
          temp.innerHTML = html;
          return temp.querySelector(".wishlist-item");
        } catch {
          return null;
        }
      }),
    );

    container.classList.remove("is-loading");

    const fragment = document.createDocumentFragment();
    cards.forEach((item) => {
      if (item) fragment.appendChild(item);
    });

    if (!fragment.childNodes.length) {
      showEmpty(container);
      return;
    }

    container.appendChild(fragment);
    updateIcons(list);
  };

  const toggleHandle = async (handle) => {
    if (!handle) return;
    const current = readLocal();
    const next = current.includes(handle)
      ? current.filter((item) => item !== handle)
      : [...current, handle];
    await persist(next);
  };

  const removeHandle = async (handle, removeBtn) => {
    if (!handle) return;
    const next = await persist(readLocal().filter((item) => item !== handle));
    const item =
      removeBtn.closest(".wishlist-item") ||
      removeBtn.closest(".card-wrapper")?.parentElement;
    item?.remove();

    const container = document.getElementById("wishlist-container");
    if (container && !container.querySelector(".wishlist-item") && !next.length) {
      showEmpty(container);
    }
  };

  document.body.addEventListener("click", (event) => {
    const removeBtn = event.target.closest(".wishlist-remove");
    if (removeBtn) {
      event.preventDefault();
      removeHandle((removeBtn.dataset.handle || "").trim(), removeBtn);
      return;
    }

    const btn = event.target.closest(".wishlist-btn");
    if (!btn) return;
    event.preventDefault();
    toggleHandle((btn.dataset.handle || "").trim());
  });

  const init = async () => {
    const localHandles = readLocal();
    updateIcons(localHandles);

    const remoteHandles = await fetchRemote();
    const merged = customerId()
      ? mergeWishlists(localHandles, remoteHandles)
      : localHandles;

    if (customerId() && merged.join(",") !== remoteHandles.join(",")) {
      await persist(merged);
    } else {
      writeLocal(merged);
      updateIcons(merged);
    }

    await loadWishlistPage(merged);
  };

  if (document.readyState === "loading") {
    document.addEventListener("DOMContentLoaded", init);
  } else {
    init();
  }
})();

Step 14 — Create the Wishlist page in Shopify admin

  1. Online Store → Pages → Add page
  2. Title: Wishlist
  3. Theme template: wishlist (page.wishlist)
  4. Handle: wishlist so the URL is /pages/wishlist
  5. Save
  6. Select that URL in the header setting

APIs

MethodURLPurpose
POSThttps://personalize.salty.co.in/api/wishlistSave or clear
GEThttps://personalize.salty.co.in/api/wishlist-list?customerId=IDList handles

Save body:

POST /api/wishlist 
{
  "actionType": "save",
  "customerId": "1234567890",
  "productHandle": "serum-handle,face-wash-handle"
}

To clear the list, send "actionType": "remove".

Test checklist

  1. Logged out: click a collection heart → it fills and the header count becomes 1
  2. /pages/wishlist shows that product; X removes it
  3. Log in and save a product
  4. Incognito + log in → list loads from the server
  5. Save as guest, then log in → lists merge
  6. PDP heart toggles
  7. App admin → Wishlist tab shows the customer
  8. Network: personalize.salty.co.in/api/wishlist returns 200, no CORS error
